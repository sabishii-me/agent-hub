    pub fn is_running(&self, sid: &str) -> bool {
        if !self.running.lock().expect("running").contains_key(sid) {
            return false;
        }
        self.requests
            .lock()
            .expect("requests")
            .get(sid)
            .map(|h| h.is_alive())
            .unwrap_or(false)
    }
